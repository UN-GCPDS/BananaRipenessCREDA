\section{Visión por Computador: ¿Qué y para qué?}

% 3. DEFINICIÓN
\begin{frame}{¿Qué es Visión por Computador?}
    \begin{itemize}
        \item Es la disciplina que busca emular la capacidad de la visión humana mediante sistemas computacionales.
        \item \textbf{Preguntas clave que resolvemos:}
        \begin{itemize}
            \item ¿Qué objetos hay en la imagen? (Clasificación).
            \item ¿Dónde están? (Localización/Detección).
            \item ¿A qué píxel pertenecen? (Segmentación).
        \end{itemize}
    \end{itemize}
    \begin{exampleblock}{Estado del Arte}
        Hoy en día, la Visión por Computador está impulsada casi en su totalidad por arquitecturas de \textbf{Deep Learning}.
    \end{exampleblock}
\end{frame}

% 4. APLICACIONES (Basado en PDI_Introducción.pdf)
\begin{frame}{¿Para qué sirve? Aplicaciones Reales}
    \begin{columns}
        \column{0.5\textwidth}
        \begin{itemize}
            \item \textbf{Salud:} Análisis de imágenes médicas (Rayos X, MRI).
            \item \textbf{Agro:} Conteo de animales y monitoreo de cultivos.
            \item \textbf{Deportes:} Análisis de rendimiento y trayectoria.
        \end{itemize}
        \column{0.5\textwidth}
        \begin{itemize}
            \item \textbf{Industria:} Inspección de calidad automática.
            \item \textbf{Seguridad:} Reconocimiento facial y biometría.
            \item \textbf{Autónomos:} Navegación de drones y vehículos.
        \end{itemize}
    \end{columns}
\end{frame}

\section{Estructura del Curso}

% 6. UNIDAD 1: PROCESAMIENTO DIGITAL DE IMÁGENES
\begin{frame}{Unidad 1: Procesamiento Digital de Imágenes (PDI)}
    \textbf{Fundamentos de la imagen digital:}
    \begin{itemize}
        \item Formación de la imagen y espacios de color (RGB, HSV, Lab).
        \item Operaciones puntuales y espaciales.
        \item Filtrado, detección de bordes y transformaciones morfológicas.
        \item Extracción de características tradicionales (LBP, HOG).
    \end{itemize}
    \begin{block}{Herramienta Clave}
        Dominio de \textbf{OpenCV} para pre-procesamiento de datos.
    \end{block}
\end{frame}

% 7. UNIDAD 2: REDES NEURONALES CONVOLUCIONALES
\begin{frame}{Unidad 2: Redes Neuronales Convolucionales (CNN)}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Conceptos Base:}
        \begin{itemize}
            \item Capas de Convolución y Pooling.
            \item Funciones de activación (ReLU, Softmax).
            \item Optimización y Backpropagation.
        \end{itemize}
        \column{0.5\textwidth}
        \textbf{Arquitecturas Clásicas:}
        \begin{itemize}
            \item LeNet, AlexNet, VGG.
            \item ResNet y el concepto de conexiones residuales.
            \item MobileNets para dispositivos móviles.
        \end{itemize}
    \end{columns}
\end{frame}

% 8. UNIDAD 3: TAREAS AVANZADAS DE VISIÓN
\begin{frame}{Unidad 3: Tareas de Alto Nivel}
    \begin{itemize}
        \item \textbf{Detección de Objetos:} YOLO (You Only Look Once), SSD y Faster R-CNN.
        \item \textbf{Segmentación Semántica e Instancia:} U-Net y Mask R-CNN.
        \item \textbf{Transfer Learning:} Cómo aprovechar modelos pre-entrenados en ImageNet para problemas específicos.
    \end{itemize}
    \vspace{0.3cm}
    \begin{alertblock}{Aplicación Práctica}
        Implementación de proyectos de clasificación y detección con \textbf{PyTorch}.
    \end{alertblock}
\end{frame}

% 9. UNIDAD 4: VISIÓN MODERNA Y TRANSFORMERS
\begin{frame}{Unidad 4: El Futuro de la Visión}
    \begin{itemize}
        \item \textbf{Vision Transformers (ViT):} Aplicando mecanismos de atención a imágenes.
        \item \textbf{Modelos Generativos:} Introducción a GANs y Modelos de Difusión.
        \item \textbf{MLOps para Visión:} Gestión de datasets, versionamiento de modelos y despliegue.
    \end{itemize}
\end{frame}
